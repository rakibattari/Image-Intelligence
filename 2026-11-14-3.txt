                if (hasGPS) {
                    riskBox.className = "bg-red-500/10 border border-red-500/30 rounded-3xl p-6 sm:p-8 shadow-xl";
                    riskMsg.innerHTML = "সতর্কতা! এই ছবিটিতে জিপিএস (GPS) লোকেশন ও সেন্সর মেটাডাটা যুক্ত রয়েছে। এই ছবিটি সোশ্যাল মিডিয়ায় শেয়ার করলে আপনার বর্তমান লোকেশন বা গোপনীয়তা ফাঁস হওয়ার উচ্চ ঝুঁকি রয়েছে।";
                } else {
                    riskBox.className = "bg-emerald-500/10 border border-emerald-500/30 rounded-3xl p-6 sm:p-8 shadow-xl";
                    riskMsg.innerHTML = "নিরাপদ! এই ছবিটিতে কোনো সংবেদনশীল জিপিএস লোকেশন বা ঝুঁকিপূর্ণ এক্সিফ ডেটা পাওয়া যায়নি। এটি শেয়ার করা তুলনামূলক নিরাপদ।";
                }

                const keys = Object.keys(allMetaData);
                if (keys.length === 0) {
                    tableBody.innerHTML = `<tr><td colspan="2" class="py-4 px-4 text-center text-slate-500">এই ছবিটিতে কোনো এক্সিফ (EXIF) ডেটা পাওয়া যায়নি।</td></tr>`;
                    return;
                }

                keys.forEach(key => {
                    let val = allMetaData[key];
                    if (typeof val === 'object') {
                        val = JSON.stringify(val);
                    }
                    const row = `<tr class="hover:bg-slate-800/50 transition">
                        <th class="py-3 px-4 font-semibold text-purple-300 w-1/3">${key}</th>
                        <td class="py-3 px-4 text-slate-300 w-2/3 break-all">${val}</td>
                    </tr>`;
                    tableBody.innerHTML += row;
                });
            });
        }

        // Mobile Menu Toggle
        const mobileMenuBtn = document.getElementById('mobileMenuBtn');
        const mobileMenu = document.getElementById('mobileMenu');
        if (mobileMenuBtn && mobileMenu) {
            mobileMenuBtn.addEventListener('click', () => {
                mobileMenu.classList.toggle('hidden');
            });
        }
